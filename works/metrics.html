口径：曝光 ＝ 各平台站内推荐位 + 搜索位 + 竞价广告的展示量； 加购未下单、下单未支付均计入流失，只有支付完成计入 GMV 与收入核算。 「下单→支付」不足 100% 的差额，是被取消订单与退款吃掉的量。 1 月全链路转化从 0.798% 提升到 0.951%， 提升主要来自点击→加购（31.9% → 33.4%）—— 说明 1 月的改善来自流量精准度，不是靠加曝光堆出来的。

数据说明：原业务的原始数据未随离职保留。本作品使用 同等表结构、同等字段含义、同等量级的脱敏仿真数据集， 全部指标由脚本自上而下真实计算，不使用任何硬写数值。 业务规则均按真实口径还原：GMV 只计支付完成订单、取消与退款不冲减 GMV、 库存按成本价计价且不进入利润表、备货水位按「爆款 / 常规 / 滞销」三档策略管理。

核心 SQL（6 段）

① 经营宽表（订单 × SKU × 渠道 三表关联）
-- ① 经营宽表：订单明细 / SKU / 渠道 三表关联（统一 平台·SKU·日期 口径）
WITH base AS (
SELECT o.order_id, o.pay_time, o.platform, o.category, o.sku_id,
       o.qty, o.pay_amount, o.status,
       s.list_price, s.cost_rate, s.age_days, s.stock_role,
       DATEDIFF('day', o.pay_time, '2025-12-01') AS day_index
FROM   fact_order_item o
JOIN   dim_sku    s ON s.sku_id  = o.sku_id
JOIN   dim_channel c ON c.platform = o.platform
WHERE  o.status IN ('支付完成','退款')
)
SELECT * FROM base;
② 经营分析主表：月份 × 品类 × 平台 四层拆解
-- ② 经营分析主表：月份 × 品类 × 平台 四层拆解
SELECT DATE_TRUNC('month', pay_time)                AS 月份,
       category, platform,
       SUM(qty)                                     AS 订单量,
       SUM(pay_amount)                              AS gmv,
       SUM(pay_amount) / SUM(qty)                   AS 客单价,
       SUM(pay_amount * (1 - cost_rate))            AS 毛利额,
       SUM(pay_amount * (1 - cost_rate))
         / SUM(pay_amount)                         AS 毛利率
FROM   base WHERE status = '支付完成'
GROUP  BY 1,2,3 ORDER BY 1, 毛利额 DESC;
③ 每日 GMV 与利润
-- ③ 每日 GMV 与利润：按支付日期聚合
SELECT DATE(pay_time)                            AS 日期,
       SUM(qty)                                  AS 订单量,
       SUM(pay_amount)                           AS gmv,
       SUM(pay_amount * (1 - cost_rate))         AS 毛利额
FROM   base WHERE status = '支付完成'
GROUP  BY 1 ORDER BY 毛利额 DESC;
④ 渠道漏斗：曝光 → 点击 → 加购 → 下单 → 支付完成
-- ④ 渠道漏斗：五状态全集（只有支付完成计入 GMV）
SELECT platform, DATE(event_date)                AS 日期,
       MAX(CASE WHEN stage = '曝光'     THEN cnt END) AS 曝光,
       MAX(CASE WHEN stage = '点击'     THEN cnt END) AS 点击,
       MAX(CASE WHEN stage = '加购'     THEN cnt END) AS 加购,
       MAX(CASE WHEN stage = '下单'     THEN cnt END) AS 下单,
       MAX(CASE WHEN stage = '支付完成' THEN cnt END) AS 支付完成
FROM   fact_traffic_funnel
GROUP  BY 1,2 ORDER BY 1,2;
⑤ 库存快照：期末库存 / 备货覆盖 / 滞销与缺货判定
-- ⑤ 库存快照：期末库存、备货覆盖月数与滞销/缺货判定
SELECT s.sku_id, s.category, s.stock_role, s.age_days,
       MAX(i.end_stock)                          AS 期末库存,
       MAX(i.restock_qty)                        AS 本期补货,
       MAX(i.sold_qty)                           AS 本期售出,
       MAX(i.end_stock) / NULLIF(MAX(i.sold_qty),0) AS 备货覆盖月数,
       CASE WHEN MAX(i.end_stock) / NULLIF(MAX(i.sold_qty),0) > 1.26
             AND s.age_days > 100 THEN '滞销'
            WHEN MAX(i.end_stock) / NULLIF(MAX(i.sold_qty),0) < 0.75
             THEN '缺货风险' ELSE '健康' END      AS 库存判定
FROM   fact_inventory_snapshot i
JOIN   dim_sku s ON s.sku_id = i.sku_id
GROUP  BY 1,2,3,4 ORDER BY 期末库存 DESC;
⑥ 资金占用率与周转天数
-- ⑥ 资金占用率与库存周转天数
SELECT DATE_TRUNC('month', snap_date)             AS 月份,
       SUM(end_stock * list_price * cost_rate)   AS 期末库存金额,
       SUM(end_stock * list_price * cost_rate)
         / SUM(DISTINCT snap_date)               AS 日均库存,
       SUM(end_stock * list_price * cost_rate)
         / (SUM(sold_qty * list_price * cost_rate) / 30) AS 库存周转天数
FROM   fact_inventory_snapshot i
JOIN   dim_sku s ON s.sku_id = i.sku_id
GROUP  BY 1 ORDER BY 1;

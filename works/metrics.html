提前期（Lead Time）结构：三段定价的依据订单维度

Lead Time ＝ 收单日期距起飞日期的天数，分三段：≤3 天（临期捡漏 / 商旅刚需）、 3–7 天（常规决策期）、>7 天（提前规划 / 早鸟）。三段需求弹性完全不同，理应分开定价。

核心 SQL（7 段）

① 经营宽表（四表关连）
-- ① 经营宽表：航班 / 订单 / 销售 / 退改 四表关连（统一店铺·SKU·日期·起飞维度口径）
WITH base AS (
  SELECT f.flight_no, f.route_code, f.airline_code, f.flight_date, f.takeoff_time,
         o.platform, o.site, o.order_id, o.order_date, o.ticket_seq,
         o.unit_cost, o.unit_price, o.pax_gender, o.pax_age, o.pax_nationality,
         DATEDIFF('day', o.order_date, f.flight_date) AS lead_days,
         o.status
  FROM   ods_flight_info  f
  JOIN   ods_order_ticket o ON o.flight_no = f.flight_no AND o.flight_date = f.flight_date
  LEFT JOIN ods_refund      r ON r.order_id = o.order_id AND r.ticket_seq = o.ticket_seq
)
SELECT * FROM base WHERE status IN ('出票完成','退票');
② 经营分析：月度 × 方向 × 航司 × 航线 四层拆解
-- ② 经营分析主表：起飞维度 月度 × 航线方向 × 航司 × 航线 四层拆解
SELECT DATE_TRUNC('month', flight_date)                    AS 起飞月,
       region_code, airline_code, route_code,
       COUNT(*)                                            AS 出票数,
       COUNT(DISTINCT order_id)                            AS 订单数,
       SUM(unit_price)                                     AS gmv,
       ROUND(SUM(unit_price - unit_cost) / COUNT(*), 2)    AS 单票利润,
       SUM(unit_price - unit_cost)                         AS 毛利
FROM   base WHERE status = '出票完成'
GROUP  BY 1,2,3,4 ORDER BY 1, 毛利 DESC;
③ 每日利润（起飞维度 / 订单维度）
-- ③ 每日利润：起飞维度按航班日期 / 订单维度按收单日期
SELECT flight_date AS 起飞日, COUNT(*) AS 出票数,
       SUM(unit_price) AS gmv, SUM(unit_price - unit_cost) AS 利润
FROM   base WHERE status = '出票完成'
GROUP  BY 1 ORDER BY 利润 DESC;
-- 订单维度把 flight_date 换成 order_date 即可
④ 平台侧：搜索 → 点击 → 出票 → 利润
-- ④ 平台侧：搜索（航线级）与点击 / 出票（航班级）按 平台×日期×航线 对齐
WITH s AS (SELECT search_date, platform, route_code, SUM(search_volume) AS sv
           FROM ods_platform_search GROUP BY 1,2,3),
     c AS (SELECT order_date, platform, route_code, COUNT(*) AS clicks,
                  SUM(CASE WHEN status = '出票完成' THEN 1 ELSE 0 END) AS done,
                  SUM(unit_price - unit_cost) AS profit
           FROM base WHERE status = '出票完成' GROUP BY 1,2,3)
SELECT c.platform, SUM(s.sv) AS 搜索量, SUM(c.clicks) AS 点击,
       SUM(c.clicks) / SUM(s.sv) AS 搜索到点击,
       SUM(c.done) / SUM(c.clicks) AS 点击到出票, SUM(c.profit) AS 利润
FROM   c JOIN s ON s.platform = c.platform AND s.route_code = c.route_code
                AND s.search_date = c.order_date
GROUP  BY 1 ORDER BY 利润 DESC;
⑤ 流量漏斗：五状态全集
-- ⑤ 流量漏斗：五状态全集（出票完成才算收益）
SELECT DATE_TRUNC('month', order_date)                    AS 收单月,
       COUNT(*)                                            AS 点击,
       SUM(CASE WHEN status IN ('待支付未付款','出票完成','退票')
                THEN 1 ELSE 0 END)                         AS 待支付,
       SUM(CASE WHEN status IN ('出票完成','退票')
                THEN 1 ELSE 0 END)                         AS 支付,
       SUM(CASE WHEN status = '出票完成' THEN 1 ELSE 0 END) AS 出票完成
FROM   base GROUP BY 1 ORDER BY 1;
⑥ 每日库存与损位
-- ⑥ 每日库存与损位：快照表按 航班×快照日期 还原起飞时配额与损位
SELECT flight_date, COUNT(DISTINCT flight_no)              AS 当日航班数,
       MAX(quota_seats)                                   AS 配额,
       MAX(cum_sold_seats)                                AS 售出,
       MAX(quota_seats) - MAX(cum_sold_seats)             AS 损位,
       MAX(cum_sold_seats) / MAX(quota_seats)             AS 动销率,
       SUM(CASE WHEN is_add_quota_day = 1
                THEN added_seats ELSE 0 END)              AS 加舱座位
FROM   dwd_inventory_daily GROUP BY 1 ORDER BY 损位 DESC;
⑦ 航班级损位排行
-- ⑦ 航班级损位排行：定位具体是哪些航班压了库存
SELECT f.flight_no, f.route_code, f.flight_date, f.takeoff_time,
       MAX(d.quota_seats)                        AS 配额,
       MAX(d.cum_sold_seats)                     AS 售出,
       MAX(d.quota_seats) - MAX(d.cum_sold_seats) AS 损位,
       MAX(d.cum_sold_seats) / MAX(d.quota_seats) AS 动销率
FROM   dwd_inventory_daily d
JOIN   dim_flight f USING (flight_no, flight_date)
GROUP  BY 1,2,3,4 ORDER BY 损位 DESC LIMIT 20;
